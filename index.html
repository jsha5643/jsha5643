ESP32 MicroPython 수신부 펌웨어 연동 가이드

마이크로컨트롤러(ESP32)에서 Nordic UART Service를 오픈하고, 브라우저가 보낸 문자열(1, 2, 3, 4, 5)을 수신해 GPIO 14(빨강), 27(초록), 26(파랑) 핀에 연결된 RGB 꼬마 전구를 제어하는 MicroPython 소스코드입니다.

main.py (ESP32 MicroPython) 
# ESP32 MicroPython BLE UART Receiver & RGB LED controller
import bluetooth
import machine
import time

class BLEUART:
    def __init__(self, ble, name="ESP_UART"):
        self.ble = ble
        self.ble.active(True)
        self.ble.irq(self._irq)
        self.name = name
        self.connections = set()
        self.rx_callback = None
        self._register()
        self._advertise()

    def _irq(self, event, data):
        if event == 1: # Connect
            conn_handle, _, _ = data
            self.connections.add(conn_handle)
            print("BLE connected")
        elif event == 2: # Disconnect
            conn_handle, _, _ = data
            if conn_handle in self.connections:
                self.connections.remove(conn_handle)
            print("BLE disconnected")
            self._advertise()
        elif event == 3: # GATTS Write
            conn_handle, value_handle = data
            if value_handle == self.rx_handle:
                data_bytes = self.ble.gatts_read(self.rx_handle)
                if self.rx_callback:
                    self.rx_callback(data_bytes)

    def _register(self):
        UART_UUID = bluetooth.UUID("6e400001-b5a3-f393-e0a9-e50e24dcca9e")
        TX_UUID = bluetooth.UUID("6e400003-b5a3-f393-e0a9-e50e24dcca9e")
        RX_UUID = bluetooth.UUID("6e400002-b5a3-f393-e0a9-e50e24dcca9e")
        
        UART_SERVICE = (
            UART_UUID,
            (
                (TX_UUID, bluetooth.FLAG_NOTIFY),
                (RX_UUID, bluetooth.FLAG_WRITE | bluetooth.FLAG_WRITE_NO_RESPONSE),
            ),
        )
        ((self.tx_handle, self.rx_handle),) = self.ble.gatts_register_services((UART_SERVICE,))

    def _advertise(self):
        payload = bytearray(b"\x02\x01\x06")
        name_bytes = self.name.encode("utf-8")
        payload += bytearray([len(name_bytes) + 1, 0x09]) + name_bytes
        uuid_bytes = bluetooth.UUID("6e400001-b5a3-f393-e0a9-e50e24dcca9e").bin
        payload += bytearray([len(uuid_bytes) + 1, 0x07]) + uuid_bytes
        self.ble.gap_advertise(100000, payload)

    def irq(self, handler):
        self.rx_callback = handler

# RGB LED GPIO Pins (보드 구성에 맞게 핀 번호를 변경해 주세요)
# Common Cathode (공통 음극) RGB LED 기준
RED_PIN = 14
GREEN_PIN = 27
BLUE_PIN = 26

r_pin = machine.Pin(RED_PIN, machine.Pin.OUT)
g_pin = machine.Pin(GREEN_PIN, machine.Pin.OUT)
b_pin = machine.Pin(BLUE_PIN, machine.Pin.OUT)

# RGB LED 끄기
def clear_rgb():
    # Common Cathode인 경우 0이 OFF, Common Anode인 경우 1이 OFF입니다.
    r_pin.value(0)
    g_pin.value(0)
    b_pin.value(0)

clear_rgb()

# 수신된 명령어 처리
def on_rx(data_bytes):
    try:
        cmd = data_bytes.decode("utf-8").strip()
        print("Received Command:", cmd)
        
        # 1: 빨강 (Red)
        if cmd == "1":
            r_pin.value(1)
            g_pin.value(0)
            b_pin.value(0)
        # 2: 초록 (Green)
        elif cmd == "2":
            r_pin.value(0)
            g_pin.value(1)
            b_pin.value(0)
        # 3: 파랑 (Blue)
        elif cmd == "3":
            r_pin.value(0)
            g_pin.value(0)
            b_pin.value(1)
        # 4: 보라 (Purple)
        elif cmd == "4":
            r_pin.value(1)
            g_pin.value(0)
            b_pin.value(1)
        # 5: 꺼짐 (Off)
        elif cmd == "5":
            clear_rgb()
        else:
            # 매핑 안된 값 수신 시 소등
            clear_rgb()
            return
            
    except Exception as e:
        print("Command error:", e)

# Main
ble = bluetooth.BLE()
uart = BLEUART(ble, name="ESP_UART")
uart.irq(on_rx)
print("ESP32 BLE RGB LED receiver is running. Advertising...")
